WITH latest AS (
    SELECT run_id FROM asnf_monitoring.flags ORDER BY detected_at DESC LIMIT 1
),
rules (rule_id, rule_no, rule_name) AS (
    VALUES
        ('GRF06', '100', 'Provider/Patient Address Match'),
        ('GRF07', '101', 'Provider/Patient Name Match'),
        ('GRF08', '102', 'Provider/Patient Phone Match'),
        ('GEO02', '103', 'Institutional/Correctional Address')
),
hits AS (
    SELECT f.rule_id, f.case_id
    FROM asnf_monitoring.flags f
    JOIN latest l
        ON l.run_id = f.run_id
    LEFT JOIN asnfdm.pcd_phi_consolidated_vw t
        ON t.case_id = f.case_id
    WHERE f.rule_id IN ('GRF06', 'GRF07', 'GRF08', 'GEO02')
      -- Optional source filter: uncomment ONE of these
      -- AND upper(btrim(t.external_source)) = 'SNX'
      -- AND upper(btrim(t.external_source)) = 'REGALORX'
      -- AND upper(btrim(t.external_source)) IN ('SNX', 'REGALORX')
)
SELECT
    r.rule_no                    AS "Rule No",
    r.rule_name                  AS "Rule",
    COUNT(DISTINCT h.case_id)    AS "Count"
FROM rules r
LEFT JOIN hits h
    ON h.rule_id = r.rule_id
GROUP BY r.rule_no, r.rule_name
ORDER BY r.rule_no;
